\begin{enumerate}
\setcounter{enumi}{0}
\item Formulas to know
\begin{enumerate}
\item[(a)] Derivative and graph of $\sin^{-1}(x)$.

\item[(b)] State the Inverse Function Theorem.

\end{enumerate}

\setcounter{enumi}{1}
\item Logarithmic differentiation
\begin{enumerate}
\item[(a)] Find $y'$ if $y=x^{\sqrt{x}}$.

\item[(b)] Find $y'$ if $y=(\sin x)^{\sin x}$.

\item[(c)] Find $y'(0)$ if $y=\frac{(x+1)^4e^x}{\sqrt{1+3x}}$.

\end{enumerate}

\setcounter{enumi}{2}
\item Derivatives of logarithms
\begin{enumerate}
\item[(a)] Find $y$ if $y'=\frac{\sin(x)}{\cos(x)}$.

\item[(b)] Find the domain and range of $f(x)=\ln(\sqrt{9-e^x})$.

\item[(c)] Let $f(x)=3-e^{-x}$. Find the domain and range of $f$, find $f^{-1}(x)$, and state the domain and range of $f^{-1}$. Does the graph of $f$ have a horizontal asymptote?

\item[(d)] Find $y'(1)$ if $y=3^x\log_3(x)$.

\end{enumerate}

\setcounter{enumi}{3}
\item Inverse trig functions
\begin{enumerate}
\item[(a)] Prove the identity $\frac{d}{dx}\sec^{-1}(x)=\frac{1}{x\sqrt{x^2-1}}$ (ignore domain issues).

\item[(b)] Find $y$ if $y'=\frac{x^2}{\frac19+x^6}$.

\item[(c)] Find the equation of the tangent line to $y=2\tan^{-1}(x^2)$ at $x=1$.

\item[(d)] Write $\cos(\tan^{-1}(x/2))$ as an algebraic function of $x$.

\item[(e)] Find $y$ if $y'=\frac{x}{\sqrt{1-x^4}}$.

\end{enumerate}

\setcounter{enumi}{4}
\item Hyperbolic functions
\begin{enumerate}
\item[(a)] Find $\cosh^{-1}(2)$.

\item[(b)] Evaluate $\lim_{x\to\infty}\coth x$.

\item[(c)] Prove that the derivative of $\tanh x$ is $\operatorname{sech}^2x$.

\item[(d)] Find $y'$ if $y=\tan^{-1}(\sinh x)$, and simplify.

\item[(e)] Evaluate $\int_{-\ln3}^{\ln3}\cosh x\,dx$.

\end{enumerate}

\setcounter{enumi}{5}
\item Inverse Function Theorem
\begin{enumerate}
\item[(a)] Show $f(x)=x+x^3+2x^5$ is 1-1, then find the derivative of $f^{-1}$ at $(4,f^{-1}(4))$.

\item[(b)] $f(x)=5x+2\sin x+2\cos x$, at $(2,f^{-1}(2))$.

\item[(c)] Let $f(x)=x+e^x$. Find the equation of the tangent line to the graph of $f^{-1}$ at $x=1$.

\item[(d)] Let $f(x)=2x^3+\ln x+\sin(\pi x)$ for $x>0$ (assume $f$ is 1-1). Find the slope of the tangent line to the graph of $f^{-1}$ at $x=2$.

\end{enumerate}

\setcounter{enumi}{6}
\item L'H\^opital's rule
\begin{enumerate}
\item[(a)] $\lim_{x\to0^+}x^x$

\item[(b)] $\lim_{x\to0^+}(1+\sin(2x))^{\cot x}$

\item[(c)] $\lim_{x\to0^+}\left(\frac1x-\frac{1}{e^x-1}\right)$

\item[(d)] $\lim_{x\to1^-}\frac{\ln x}{\cos^{-1}(x)}$

\item[(e)] $\lim_{x\to\infty}x\left(\frac{\pi}{2}-\tan^{-1}(2x)\right)$

\item[(f)] $\lim_{x\to\infty}\sin^{-1}\left(\frac{e^x}{2e^x+1}\right)$

\item[(g)] $\lim_{x\to\infty}\frac{\cosh(\ln x)}{x}$

\end{enumerate}

\setcounter{enumi}{7}
\item Substitution
\begin{enumerate}
\item[(a)] $\int_0^{(\pi/2)^{1/3}}x^2\sin x^3\,dx$

\item[(b)] $\int_{\pi/4}^{\pi/2}\cot x\,dx$

\item[(c)] $\int_1^e\frac{(\ln x)^2}{x}\,dx$

\item[(d)] $\int\frac{\cos x}{4+\sin^2x}\,dx$

\item[(e)] $\int\frac{dx}{\sqrt x\,(1+\sqrt x)}$

\end{enumerate}

\setcounter{enumi}{8}
\item Integration by parts
\begin{enumerate}
\item[(a)] $\int_1^e x^4\ln(x)\,dx$

\item[(b)] $\int_0^\pi e^x\sin(x)\,dx$

\item[(c)] $\int x\sec^2(x)\,dx$

\item[(d)] $\int x\sin(2x)\,dx$

\item[(e)] $\int_0^1x\tan^{-1}(x)\,dx$

\end{enumerate}

\setcounter{enumi}{9}
\item Trig integrals
\begin{enumerate}
\item[(a)] $\int\sin^3(x)\cos^2(x)\,dx$

\item[(b)] $\int\tan^2(x)\,dx$

\item[(c)] $\int\tan^2(x)\sec^4(x)\,dx$

\item[(d)] $\int\sin(3x)\cos(2x)\,dx$

\item[(e)] $\int_0^{\pi/3}\tan(x)\sec^3(x)\,dx$

\end{enumerate}

\setcounter{enumi}{10}
\item Trig substitution
\begin{enumerate}
\item[(a)] $\int_2^3\frac{dx}{(x^2-1)^{3/2}}$

\item[(b)] $\int\frac{dx}{(6-8x-x^2)^{3/2}}$

\item[(c)] $\int_0^{\sqrt3}\frac{x^3}{\sqrt{x^2+1}}\,dx$

\item[(d)] $\int_{\sqrt2}^2\frac{dx}{x^2\sqrt{x^2-1}}$

\item[(e)] $\int_0^1\frac{dx}{(1+x^2)^2}$

\end{enumerate}

\setcounter{enumi}{11}
\item Partial fractions
\begin{enumerate}
\item[(a)] $\int\frac{3x^2+6x+2}{x^2+3x+2}\,dx$

\item[(b)] $\int\frac{x+4}{x^3-4x}\,dx$

\item[(c)] $\int\frac{3x^2-x+8}{x^3+4x}\,dx$

\item[(d)] Set up, but do not solve, the partial fraction decomposition of $\frac{x+1}{x^2(x-3)(x^2+1)^2}$.

\end{enumerate}

\setcounter{enumi}{12}
\item Improper integrals
\begin{enumerate}
\item[(a)] $\int_0^1\frac{\ln x}{x}\,dx$

\item[(b)] $\int_1^\infty\frac{dx}{x^p}$ (consider $p\le1$ and $p>1$)

\item[(c)] $\int_0^\infty e^{x^2}\,dx$ (don't determine the value)

\item[(d)] $\int_0^\infty\frac{e^x}{(1+e^x)^3}\,dx$

\item[(e)] $\int_1^\infty\frac{\ln x}{x^3}\,dx$

\item[(f)] $\int_0^2\frac{dx}{(x-1)^{2/3}}$

\item[(g)] $\int_2^\infty\frac{dx}{4+x^2}$

\end{enumerate}

\setcounter{enumi}{13}
\item Numerical integration
\begin{enumerate}
\item[(a)] $\int_0^1(x^3-3x^2+2x+1)\,dx$: over or under estimates, then $M_4$ and $T_4$.

\item[(b)] Use the trapezoid rule with $n=3$ to approximate $\int_0^32^x\,dx$ in exact form.

\item[(c)] How large must $n$ be so that the trapezoid rule approximates $\int_1^2\frac{dx}{x}$ with error less than $10^{-4}$?

\end{enumerate}

\setcounter{enumi}{14}
\item Sequences
\begin{enumerate}
\item[(a)] If an alternating sequence converges, what must it converge to?

\item[(b)] Write $\{-\frac18,\frac1{27},-\frac1{64},\frac1{125},\dots\}$ as a function of $n$.

\item[(c)] Monotonic? Bounded? $a_n=1+\frac{(-1)^n}{n}$ and $b_n=\frac{n}{n+1}$ ($n\ge1$).

\item[(d)] Show that $\lim_{n\to\infty}\frac{3^n}{n!}=0$.

\end{enumerate}

\setcounter{enumi}{15}
\item Test for Divergence (TFD)
\begin{enumerate}
\item[(a)] $\sum_{n=1}^\infty\frac{(-1)^nn}{n+1}$

\item[(b)] $\sum_{n=0}^\infty\frac{n^2+1}{6n^2+6n+1}$

\item[(c)] Can you use the TFD on $\sum_{n=2}^\infty\frac{1}{n^2}$?

\item[(d)] $\sum_{n=1}^\infty2^{1/n}$

\item[(e)] Suppose $b_n>0$ and $\sum b_n$ converges. What does the TFD say about $\sum\frac{1}{b_n}$?

\end{enumerate}

\setcounter{enumi}{16}
\item Telescoping series
\begin{enumerate}
\item[(a)] $\sum_{n=2}^\infty\frac{1}{n^2-1}$

\item[(b)] $\sum_{n=1}^\infty\left[\tan^{-1}(n+2)-\tan^{-1}(n)\right]$

\item[(c)] $\sum_{n=1}^\infty\left[f(n)-f(n+1)\right]$, where $\lim_{n\to\infty}f(n)=L$

\item[(d)] $\sum_{n=1}^\infty\left[f(n)-f(n+k)\right]$, $k$ a positive integer, $\lim f(n)=L$

\item[(e)] Determine if $\sum_{n=1}^\infty\ln\left(1+\frac1n\right)$ converges or diverges by writing out its partial sums.

\end{enumerate}

\setcounter{enumi}{17}
\item Geometric series
\begin{enumerate}
\item[(a)] $\sum_{n=0}^\infty\frac{4^{2n-1}}{17^n}$

\item[(b)] $\sum_{n=-4}^\infty\left(\frac12\right)^n$

\item[(c)] $\sum_{n=1}^\infty\frac{3^{n+1}}{4^{n-1}}$

\item[(d)] $2-1+\frac12-\frac14+\frac18-\cdots$

\item[(e)] For which $a$ does $\sum_{k=1}^\infty\left(\frac{2a}{1+a^2}\right)^k$ converge? Find its value.

\end{enumerate}

\setcounter{enumi}{18}
\item Integral Test
\begin{enumerate}
\item[(a)] $\sum_{n=2}^\infty\frac{1}{n(\ln n)^2}$

\item[(b)] Show $\sum_{n=1}^\infty\frac{1}{n^p}$ converges if $p>1$ and diverges if $0<p\le1$.

\item[(c)] $\sum_{n=1}^\infty ne^{-n^2}$

\item[(d)] Can you use the integral test on $\sum_{n=1}^\infty\frac{2+\sin n}{n^2}$?

\end{enumerate}

\setcounter{enumi}{19}
\item p-series
\begin{enumerate}
\item[(a)] $\sum_{n=5}^\infty\frac{1}{n^e}$

\item[(b)] $\sum_{n=1}^\infty\frac{1}{n^{0.000001}}$

\item[(c)] $\sum_{n=1}^\infty\frac{n^2+\sqrt n}{n^4}$

\item[(d)] For which $p$ does $\sum_{n=1}^\infty\frac{1}{n^{2p-1}}$ converge?

\end{enumerate}

\setcounter{enumi}{20}
\item Direct Comparison Test (DCT) and Limit Comparison Test (LCT)
\begin{enumerate}
\item[(a)] $\sum_{n=1}^\infty\sin(1/n)$

\item[(b)] $\sum_{n=0}^\infty\frac{\tan^{-1}(n)}{n^2+1}$

\item[(c)] $\sum_{n=1}^\infty\frac{1}{2^n+n}$

\item[(d)] $\sum_{n=1}^\infty\frac{n^2-2n}{n^{7/2}+1}$

\item[(e)] $\sum_{n=1}^\infty\frac{1}{n^2+n\cos n}$

\end{enumerate}

\setcounter{enumi}{21}
\item Alternating Series Test (AST)
\begin{enumerate}
\item[(a)] $\sum_{n=2}^\infty\frac{(-1)^n}{n\ln n}$

\item[(b)] $\sum_{i=1}^\infty\frac{\cos(\pi i)}{i^{0.1}}$

\item[(c)] $\sum_{k=2}^\infty\frac{(-1)^{k+15}}{\ln k}$

\item[(d)] $\sum_{n=1}^\infty\frac{(-1)^n\ln n}{\sqrt n}$

\item[(e)] $\sum_{n=1}^\infty\frac{(-1)^n\sqrt n}{n+4}$

\item[(f)] $a_n=\frac1n$ for $n$ odd, $a_n=\frac1{n^2}$ for $n$ even. Can you use the AST on $\sum(-1)^na_n$?

\end{enumerate}

\setcounter{enumi}{22}
\item Ratio Test
\begin{enumerate}
\item[(a)] $\sum_{n=0}^\infty\frac{n!}{n^n}$

\item[(b)] $\sum_{n=0}^\infty\frac{(2n+1)!}{3^n}$

\item[(c)] $\sum_{n=1}^\infty\frac{n^24^n}{n!}$

\end{enumerate}

\setcounter{enumi}{23}
\item Root Test
\begin{enumerate}
\item[(a)] $\sum_{n=1}^\infty\left(\frac{2n+1}{3n-1}\right)^n$

\item[(b)] $\sum_{n=1}^\infty\frac{n^n}{2^{n^2}}$

\item[(c)] Show the Root Test is inconclusive for $\sum\frac1n$ and $\sum\frac1{n^2}$. Which converges?

\item[(d)] $\sum_{n=1}^\infty\frac{2^n}{n^5}$: Root Test? Ratio Test?

\end{enumerate}

\setcounter{enumi}{24}
\item Absolute and conditional convergence
\begin{enumerate}
\item[(a)] $\sum_{n=1}^\infty\frac{(-1)^n}{n}$

\item[(b)] $\sum_{k=1}^\infty\frac{k\cos(\pi k)}{k^3+1}$

\item[(c)] $\sum_{n=1}^\infty\frac{(-1)^nn^3}{2^n}$

\item[(d)] $\sum_{n=1}^\infty\frac{(-1)^nn}{n^2+1}$

\item[(e)] Suppose $\sum a_n$ converges to $2$. What is $\lim a_n$? Must $\sum|a_n|$ converge?

\end{enumerate}

\setcounter{enumi}{25}
\item Radius of Convergence (ROC) and Interval of Convergence (IOC)
\begin{enumerate}
\item[(a)] $\sum_{n=0}^\infty n^nx^n$

\item[(b)] $\sum_{n=1}^\infty\frac{(-1)^nx^{2n}}{3^n(2n+1)}$

\item[(c)] $\sum_{n=1}^\infty\frac{(x+2)^n}{n\,4^n}$

\item[(d)] Center, ROC, IOC: $\sum_{n=1}^\infty\frac{2^n(x-3)^n}{n^2}$

\end{enumerate}

\setcounter{enumi}{26}
\item Power series from geometric series
\begin{enumerate}
\item[(a)] $g(x)=\frac{1}{1+x}$, $a=1$. State the ROC and IOC.

\item[(b)] $g(x)=\ln(x)$, $a=1$. State the ROC.

\item[(c)] Find the function that $\sum_{n=2}^\infty2n(n-1)x^n$ represents.

\item[(d)] $f(x)=\frac{x}{(1-x^2)^2}$, $a=0$.

\item[(e)] Find the exact value of $\sum_{n=1}^\infty\frac{n}{3^n}$.

\end{enumerate}

\setcounter{enumi}{27}
\item Taylor and Maclaurin series
\begin{enumerate}
\item[(a)] Find the Maclaurin series for $\cosh(x)$ and $\sinh(x)$.

\item[(b)] Degree 2 Taylor polynomial of $\sqrt x$ at $a=9$.

\item[(c)] Degree 3 Taylor polynomial of $\cos x$ at $a=\frac{\pi}{3}$.

\item[(d)] Maclaurin series for $F(x)=\int_0^xe^{-t^2}\,dt$ and its ROC.

\item[(e)] Find the exact value of $1-\frac{\pi^2}{3^2\cdot2!}+\frac{\pi^4}{3^4\cdot4!}-\cdots$

\item[(f)] Find the exact value of $\sum_{n=0}^\infty\frac{(\ln3)^n}{n!}$.

\item[(g)] Find the exact value of $\sum_{n=1}^\infty\frac{(-1)^{n+1}}{n\,2^n}$.

\item[(h)] Use a Maclaurin series to evaluate $\lim_{x\to0}\frac{\sin x-x}{x^3}$.

\end{enumerate}

\setcounter{enumi}{28}
\item Approximations and error bounds
\begin{enumerate}
\item[(a)] Approximate $\int_0^{0.1}\frac{dx}{1+x^4}$ to within $10^{-6}$.

\item[(b)] Use $p_3(x)$ to approximate $\tan^{-1}(x)$ near $0$, and bound the error at $x=\frac12$ with ASE.

\item[(c)] Bound the same error with Taylor's Inequality, bounding $|f^{(4)}(x)|$ on $[0,\frac12]$, where $f(x)=\tan^{-1}(x)$.

\item[(d)] $f(x)=\sum_{n=2}^\infty5n(n-1)x^n$: parts (i) to (vi).

\item[(e)] How many terms of $\sum_{n=0}^\infty\frac{(-1)^n}{n!}$ are needed to approximate its sum to within $0.001$?

\item[(f)] $f(x)=e^x$: find $T_3(x)$ and bound $|f(x)-T_3(x)|$ for $-1\le x\le1$.

\item[(g)] Approximate $\int_0^1e^{-x^2}\,dx$ to within $0.01$ (use ASE).

\end{enumerate}

\setcounter{enumi}{29}
\item Complex numbers
\begin{enumerate}
\item[(a)] Compute $(1+i)^{10}$ in $a+bi$ form.

\item[(b)] Compute $(i-1)^{12}$ in $a+bi$ form.

\end{enumerate}

\setcounter{enumi}{30}
\item Differential equations
\begin{enumerate}
\item[(a)] Show that $y=Ce^{x^2}$ solves $y'=2xy$ for every constant $C$. Find the solution with $y(0)=3$.

\item[(b)] Sketch the direction field of $y'=x+y$ at the nine points with $x,y\in\{-1,0,1\}$. Along which line are the slopes $0$? Sketch the solution through $(0,0)$.

\item[(c)] Find the equilibrium solutions of $y'=y^2-4$ and decide whether nearby solutions move toward or away from each.

\item[(d)] Euler's method with $h=0.5$ for $y'=x+y$, $y(0)=1$, to approximate $y(1)$. Over or under estimate?

\item[(e)] Euler's method with $h=-1$ for $y'=y-x$, $y(1)=3$, to approximate $y(-1)$.

\item[(f)] Solve $\frac{dy}{dx}=xy^2$, $y(0)=1$. On what interval is the solution defined?

\item[(g)] Solve $\frac{dy}{dx}=e^{x-y}$, $y(0)=\ln2$.

\item[(h)] Solve $\frac{dy}{dt}=3y-6$, $y(0)=5$.

\item[(i)] Bacteria grow at a rate proportional to size, from $200$ cells to $600$ in $2$ hours. Find $P(t)$ and when $P=5400$.

\item[(j)] Logistic: $\frac{dP}{dt}=0.5P\left(1-\frac{P}{1000}\right)$, $P(0)=200$. Carrying capacity, equilibria, growth, and $P(t)$.

\end{enumerate}

\end{enumerate}
